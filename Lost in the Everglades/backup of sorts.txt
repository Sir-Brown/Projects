/*
bool progress(const string ARRAY[], const char ARR[], char map[][5], int row, int col, int& gongs)
Parameters: 3 arrays, and 3 integers
Return type: bool
Purpose: To determine whether or not the player has won the game.
*/
bool progress(const string ARRAY[], const char ARR[], char map[][5], int row, int col, int& gongs)
{
	// Determine's the player's progress
	if (map[row][col] == 'T')
	{
		showMap(map);
		cout << "\nCongratulations!! You have rescued the tourists!" << endl;
		return true;
	}
	else if (gongs == 0)
	{
		cout << "\nGame over . . .\nThe tourists have been claimed by the everglades, never to return." << endl;
		return true;
	}
	else
	{
		return false;
	}
}
